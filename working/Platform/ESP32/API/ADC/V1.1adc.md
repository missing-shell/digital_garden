```c++
#include <stdio.h>                   // 包含标准输入输出库
#include <stdlib.h>                  // 包含标准库
#include <string.h>                  // 包含字符串处理库
#include "freertos/FreeRTOS.h"       // 包含FreeRTOS实时操作系统库
#include "freertos/task.h"           // 包含FreeRTOS任务处理库
#include "soc/soc_caps.h"            // 包含SoC功能库
#include "esp_log.h"                 // 包含ESP日志库
#include "esp_adc/adc_oneshot.h"     // 包含ESP ADC一次性读取库
#include "esp_adc/adc_cali.h"        // 包含ESP ADC校准库
#include "esp_adc/adc_cali_scheme.h" // 包含ESP ADC校准方案库

#define MYADC_CHANNEL ADC_CHANNEL_2
#define MYADC_ATTEN ADC_ATTEN_DB_11 // 定义ADC衰减值

const static char *TAG = "V1_1_ADC"; // 定义日志标签

static int adc_raw[2][10]; // 定义ADC原始数据数组
static int voltage[2][10]; // 定义电压数据数组

void app_main(void)
{
    adc_oneshot_unit_handle_t adc1_handle;
    adc_oneshot_unit_init_cfg_t init_config1 = {
        .unit_id = ADC_UNIT_1,
    };
    ESP_ERROR_CHECK(adc_oneshot_new_unit(&init_config1, &adc1_handle));

    adc_oneshot_chan_cfg_t config1 = {
        .bitwidth = ADC_BITWIDTH_DEFAULT, // 设置位宽为默认值
        .atten = MYADC_ATTEN,             // 设置衰减值
    };
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc1_handle, MYADC_CHANNEL, &config1));

    printf("maybe\n");

    adc_cali_curve_fitting_config_t cfg = {
        .unit_id = ADC_UNIT_1,
        .atten = MYADC_ATTEN,
        .bitwidth = ADC_BITWIDTH_DEFAULT,
    };

    adc_cali_handle_t handle;

    // 创建曲线拟合校准方案
    esp_err_t ret = adc_cali_create_scheme_curve_fitting(&cfg, &handle);
    if (ret == ESP_OK)
    {
        printf("\nCreated curve fitting scheme successfully\n");
    }
    else
    {
        printf("\nFailed to create curve fitting scheme\n");
        return;
    }

    // ... 在这里使用 handle 进行其他操作，比如 ADC 校准等

    // 删除曲线拟合校准方案
    adc_cali_delete_scheme_curve_fitting(handle);

    while (1)
    {
        ESP_ERROR_CHECK(adc_oneshot_read(adc1_handle, MYADC_CHANNEL, &adc_raw[0][0]));
        ESP_LOGI(TAG, "ADC%d通道[%d]原始数据： %d", ADC_UNIT_1 + 1, MYADC_CHANNEL, adc_raw[0][0]);
        vTaskDelay(pdMS_TO_TICKS(1000)); // 延迟1秒
    }
}
```
