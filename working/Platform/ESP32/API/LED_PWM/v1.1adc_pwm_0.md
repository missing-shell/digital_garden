#include <stdio.h>                   // 包含标准输入输出库
#include <stdlib.h>                  // 包含标准库
#include <string.h>                  // 包含字符串处理库
#include "freertos/FreeRTOS.h"       // 包含FreeRTOS实时操作系统库
#include "freertos/task.h"           // 包含FreeRTOS任务处理库
#include "soc/soc_caps.h"            // 包含SoC功能库
#include "esp_log.h"                 // 包含ESP日志库
#include "esp_adc/adc_oneshot.h"     // 包含ESP ADC一次性读取库
#include "esp_adc/adc_cali.h"        // 包含ESP ADC校准库
#include "esp_adc/adc_cali_scheme.h" // 包含ESP ADC校准方案库
#include "driver/ledc.h"
#include "esp_err.h"

#define MYADC_CHANNEL ADC_CHANNEL_2   //定义ADC通道
#define MYADC_ATTEN ADC_ATTEN_DB_11  // 定义ADC衰减值
#define LEDC_TIMER LEDC_TIMER_0
#define LEDC_MODE LEDC_LOW_SPEED_MODE
#define LEDC_OUTPUT_IO (18) // Define the output GPIO
#define LEDC_CHANNEL LEDC_CHANNEL_0
#define LEDC_DUTY_RES LEDC_TIMER_13_BIT // Set duty resolution to 13 bits
#define LEDC_DUTY (8192)                // Set duty to 50%. ((2 ** 13) - 1) * 50% = 4095
#define LEDC_FREQUENCY (5000)           // Frequency in Hertz. Set frequency at 5 kHz

const static char *TAG = "V1.1_LED_PWM_0"; // 定义日志标签

static int adc_raw[2][10]; // 定义ADC原始数据数组
int mapp(int x, int in_min, int in_max, int out_min, int out_max);
void ledc_init(void);

void app_main(void)
{
    // Set the LEDC peripheral configuration
    ledc_init();

    /*ADC1初始化*/
    adc_oneshot_unit_handle_t adc1_handle;
    adc_oneshot_unit_init_cfg_t init_config = {
        .unit_id = ADC_UNIT_1,
        .ulp_mode = ADC_ULP_MODE_DISABLE, // 禁用ULP模式
    };
    ESP_ERROR_CHECK(adc_oneshot_new_unit(&init_config, &adc1_handle));

    /*ADC1配置*/
    adc_oneshot_chan_cfg_t config = {
        .bitwidth = ADC_BITWIDTH_DEFAULT,
        .atten = MYADC_ATTEN,
    };
    ESP_ERROR_CHECK(adc_oneshot_config_channel(adc1_handle, MYADC_CHANNEL, &config));

    while (1)
    {
        ESP_ERROR_CHECK(adc_oneshot_read(adc1_handle, MYADC_CHANNEL, &adc_raw[0][0])); // 读取ADC1通道0的原始数据
        ESP_LOGI(TAG, "ADC%d通道[%d]原始数据： %d", ADC_UNIT_1 + 1, MYADC_CHANNEL, adc_raw[0][0]);// 打印ADC1通道0的原始数据
        
        int Duty = mapp(adc_raw[0][0], 0, 4095, 0, LEDC_DUTY);//映射
        // Set duty toadc_raw
        ESP_ERROR_CHECK(ledc_set_duty(LEDC_MODE, LEDC_CHANNEL, Duty));
        // Update duty to apply the new value
        ESP_ERROR_CHECK(ledc_update_duty(LEDC_MODE, LEDC_CHANNEL));
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

/*初始化led_pwm*/
void ledc_init(void)
{
    // Prepare and then apply the LEDC PWM timer configuration
    ledc_timer_config_t ledc_timer = {
        .speed_mode = LEDC_MODE,
        .timer_num = LEDC_TIMER,
        .duty_resolution = LEDC_DUTY_RES,
        .freq_hz = LEDC_FREQUENCY,
        .clk_cfg = LEDC_AUTO_CLK};
    ESP_ERROR_CHECK(ledc_timer_config(&ledc_timer));

    // Prepare and then apply the LEDC PWM channel configuration
    ledc_channel_config_t ledc_channel = {
        .speed_mode = LEDC_MODE,
        .channel = LEDC_CHANNEL,
        .timer_sel = LEDC_TIMER,
        .intr_type = LEDC_INTR_DISABLE,
        .gpio_num = LEDC_OUTPUT_IO,
        .duty = 0,
        .hpoint = 0
    };
    ESP_ERROR_CHECK(ledc_channel_config(&ledc_channel));
}

/*映射函数*/
int mapp(int x, int in_min, int in_max, int out_min, int out_max) {
    return (x - in_min) * (out_max - out_min) / (in_max - in_min) + out_min;
}
