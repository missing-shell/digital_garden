```c++
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/semphr.h"
#include "driver/ledc.h"
#include "esp_err.h"

#define LEDC_LS_TIMER LEDC_TIMER_1
#define LEDC_LS_MODE LEDC_LOW_SPEED_MODE
#define LEDC_LS_CH0_GPIO (8)
#define LEDC_LS_CH0_CHNANEL LEDC_CHANNEL_0
#define LEDC_LS_CH1_GPIO (5)
#define LEDC_LS_CH1_CHANNEL LEDC_CHANNEL_1
#define LEDC_LS_CH2_GPIO (0)
#define LEDC_LS_CH2_CHANNEL LEDC_CHANNEL_2
#define LEDC_LS_CH3_GPIO (1)
#define LEDC_LS_CH3_CHANNEL LEDC_CHANNEL_3

#define LEDC_TEST_CH_NUM (4)
#define LEDC_TEST_DUTY (4000)
#define LEDC_TEST_FADE_TIME (3000)

/*
 * This callback function will be called when fade operation has ended
 * Use callback only if you are aware it is being called inside an ISR
 * Otherwise, you can use a semaphore to unblock tasks
 */
static IRAM_ATTR bool cb_ledc_fade_end_event(const ledc_cb_param_t *param, void *user_arg)
{
    portBASE_TYPE taskAwoken = pdFALSE; // Define a variable to track if a task has been woken up

    if (param->event == LEDC_FADE_END_EVT) { // Check if the event is LEDC_FADE_END_EVT
        SemaphoreHandle_t counting_sem = (SemaphoreHandle_t) user_arg; // Cast the user argument as a SemaphoreHandle_t
        xSemaphoreGiveFromISR(counting_sem, &taskAwoken); // Give the semaphore from an ISR and update the taskAwoken variable
    }

    return (taskAwoken == pdTRUE); // Return the value of taskAwoken
}


void app_main(void)
{
    /*Set configuration of timers*/
    ledc_timer_config_t ledc_timer = {
        .duty_resolution = LEDC_TIMER_13_BIT, // resolution of PWM duty
        .freq_hz = 5000,                      // frequency of PWM signal
        .speed_mode = LEDC_LS_MODE,           // timer mode
        .timer_num = LEDC_LS_TIMER,            // timer index
        .clk_cfg = LEDC_AUTO_CLK,              // Auto select the source clock
    };
    ledc_timer_config(&ledc_timer); // Set configuration of timer0 for high speed channels

    /*
     * Prepare individual configuration
     * for each channel of LED Controller
     * by selecting:
     * - controller's channel number
     * - output duty cycle, set initially to 0
     * - GPIO number where LED is connected to
     * - speed mode, either high or low
     * - timer servicing selected channel
     *   Note: if different channels use one timer,
     *         then frequency and bit_num of these channels
     *         will be the same
     */
    ledc_channel_config_t ledc_channel[LEDC_TEST_CH_NUM] = {
        {
            .channel = LEDC_LS_CH0_CHNANEL,
            .duty = 0,
            .gpio_num = LEDC_LS_CH0_GPIO,
            .speed_mode = LEDC_LS_MODE,

            .hpoint = 0, // PWM信号的相位

            /*hpoint值在ESP32的LED PWM控制器（LEDC）中，
            表示PWM信号周期内的上升沿时间点。LEDC内部有一个计数器
            ，当计数器的值达到hpoint时，输出信号翻转为高电平
            。这个计数器会一直累加，直到溢出，然后再从0开始。*/

            .timer_sel = LEDC_LS_TIMER,
            .flags.output_invert =0 // 控制GPIO输出是否反转。
        },
        { 
            .channel = LEDC_LS_CH1_CHANNEL,
            .duty = 0,
            .gpio_num = LEDC_LS_CH1_GPIO,
            .speed_mode = LEDC_LS_MODE,
            .hpoint = 0,
            .timer_sel = LEDC_LS_TIMER,
            .flags.output_invert = 0
         
        },
        {
            .channel = LEDC_LS_CH2_CHANNEL,
            .duty = 0,
            .gpio_num = LEDC_LS_CH2_GPIO,
            .speed_mode = LEDC_LS_MODE,
            .hpoint = 0,
            .timer_sel = LEDC_LS_TIMER,
            .flags.output_invert =1
        }, 
        {   
            .channel = LEDC_LS_CH3_CHANNEL,
            .duty = 0,
            .gpio_num = LEDC_LS_CH3_GPIO,
            .speed_mode = LEDC_LS_MODE,
            .hpoint = 0,
            .timer_sel = LEDC_LS_TIMER,
            .flags.output_invert =1
        }
    };

    // Set LED Controller with previously prepared configuration
    for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++)
    {
        ledc_channel_config(&ledc_channel[ch]);
    }

    // Initialize fade service.
    ledc_fade_func_install(0); // Install the fade service of LEDC. Parameter 0 indicates that the interrupt is not immediately enabled after the service is installed.

    // Define callbacks structure and set the fade end event callback function.
    ledc_cbs_t callbacks = {
        .fade_cb = cb_ledc_fade_end_event // Callback function to be called when the fade operation ends.
    };

    // Create a counting semaphore.
    SemaphoreHandle_t counting_sem = xSemaphoreCreateCounting(LEDC_TEST_CH_NUM, 0); // Initial count is set to 0.

    // Iterate over LEDC channels.
    for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++)
    {
        // Register the callback function to the LEDC channel.
        // The callback function will be called when the fade operation ends, and the semaphore will be incremented.
        ledc_cb_register(ledc_channel[ch].speed_mode, ledc_channel[ch].channel, &callbacks, (void *)counting_sem); 
       
    }

    while (1) {
        printf("1. LEDC fade up to duty = %d\n", LEDC_TEST_DUTY);
        for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++) {
            ledc_set_fade_with_time(ledc_channel[ch].speed_mode,
                    ledc_channel[ch].channel, LEDC_TEST_DUTY, LEDC_TEST_FADE_TIME);
            ledc_fade_start(ledc_channel[ch].speed_mode,
                    ledc_channel[ch].channel, LEDC_FADE_NO_WAIT);
        }

        for (int i = 0; i < LEDC_TEST_CH_NUM; i++) {
            xSemaphoreTake(counting_sem, portMAX_DELAY);
        }

        printf("2. LEDC fade down to duty = 0\n");
        for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++) {
            ledc_set_fade_with_time(ledc_channel[ch].speed_mode,
                    ledc_channel[ch].channel, 0, LEDC_TEST_FADE_TIME);
            ledc_fade_start(ledc_channel[ch].speed_mode,
                    ledc_channel[ch].channel, LEDC_FADE_NO_WAIT);
        }

        for (int i = 0; i < LEDC_TEST_CH_NUM; i++) {
            xSemaphoreTake(counting_sem, portMAX_DELAY);
        }

        printf("3. LEDC set duty = %d without fade\n", LEDC_TEST_DUTY);
        for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++) {
            ledc_set_duty(ledc_channel[ch].speed_mode, ledc_channel[ch].channel, LEDC_TEST_DUTY);
            ledc_update_duty(ledc_channel[ch].speed_mode, ledc_channel[ch].channel);
        }
        vTaskDelay(1000 / portTICK_PERIOD_MS);

        printf("4. LEDC set duty = 0 without fade\n");
        for (int ch = 0; ch < LEDC_TEST_CH_NUM; ch++) {
            ledc_set_duty(ledc_channel[ch].speed_mode, ledc_channel[ch].channel, 0);
            ledc_update_duty(ledc_channel[ch].speed_mode, ledc_channel[ch].channel);
        }
        vTaskDelay(1000 / portTICK_PERIOD_MS);
    }
}
```
