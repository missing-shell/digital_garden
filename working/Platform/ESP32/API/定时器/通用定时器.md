#include <stdio.h>  // 包含标准输入输出库
#include "freertos/FreeRTOS.h"  // 包含FreeRTOS库
#include "freertos/task.h"  // 包含任务处理库
#include "freertos/queue.h"  // 包含队列处理库
#include "driver/gptimer.h"  // 包含定时器驱动库
#include "esp_log.h"  // 包含日志库

static const char *TAG = "example";  // 定义标签为"example"

typedef struct {  // 定义结构体
    uint64_t event_count;  // 事件计数
} example_queue_element_t;  // 结构体类型名

static bool IRAM_ATTR example_timer_on_alarm_cb_v1(gptimer_handle_t timer, const gptimer_alarm_event_data_t *edata, void *user_data)  // 定义定时器回调函数v1
{
    BaseType_t high_task_awoken = pdFALSE;  // 初始化任务唤醒标志为假
    QueueHandle_t queue = (QueueHandle_t)user_data;  // 获取用户数据队列
    gptimer_stop(timer);  // 停止定时器
    example_queue_element_t ele = {  // 定义队列元素
        .event_count = edata->count_value  // 获取计数值
    };
    xQueueSendFromISR(queue, &ele, &high_task_awoken);  // 从中断服务程序发送队列
    return (high_task_awoken == pdTRUE);  // 如果任务被唤醒，返回真
}

static bool IRAM_ATTR example_timer_on_alarm_cb_v2(gptimer_handle_t timer, const gptimer_alarm_event_data_t *edata, void *user_data)  // 定义定时器回调函数v2
{
    BaseType_t high_task_awoken = pdFALSE;  // 初始化任务唤醒标志为假
    QueueHandle_t queue = (QueueHandle_t)user_data;  // 获取用户数据队列
    example_queue_element_t ele = {  // 定义队列元素
        .event_count = edata->count_value  // 获取计数值
    };
    xQueueSendFromISR(queue, &ele, &high_task_awoken);  // 从中断服务程序发送队列
    return (high_task_awoken == pdTRUE);  // 如果任务被唤醒，返回真
}

static bool IRAM_ATTR example_timer_on_alarm_cb_v3(gptimer_handle_t timer, const gptimer_alarm_event_data_t *edata, void *user_data)  // 定义定时器回调函数v3
{
    BaseType_t high_task_awoken = pdFALSE;  // 初始化任务唤醒标志为假
    QueueHandle_t queue = (QueueHandle_t)user_data;  // 获取用户数据队列
    example_queue_element_t ele = {  // 定义队列元素
        .event_count = edata->count_value  // 获取计数值
    };
    xQueueSendFromISR(queue, &ele, &high_task_awoken);  // 从中断服务程序发送队列
    gptimer_alarm_config_t alarm_config = {  // 定义警报配置
        .alarm_count = edata->alarm_value + 1000000,  // 在下一秒警报
    };
    gptimer_set_alarm_action(timer, &alarm_config);  // 设置警报动作
    return (high_task_awoken == pdTRUE);  // 如果任务被唤醒，返回真
}


void app_main(void) // 主函数
{
    example_queue_element_t ele; // 定义队列元素
    QueueHandle_t queue = xQueueCreate(10, sizeof(example_queue_element_t)); // 创建一个队列
    if (!queue) { // 如果队列创建失败
        ESP_LOGE(TAG, "Creating queue failed"); // 打印错误日志
        return; // 返回
    }

    ESP_LOGI(TAG, "Create timer handle"); // 创建定时器句柄
    gptimer_handle_t gptimer = NULL; // 定义定时器句柄
    gptimer_config_t timer_config = { // 定义定时器配置
        .clk_src = GPTIMER_CLK_SRC_DEFAULT, // 默认时钟源
        .direction = GPTIMER_COUNT_UP, // 计数方向为向上
        .resolution_hz = 1000000, // 分辨率为1MHz，1 tick=1us
    };
    ESP_ERROR_CHECK(gptimer_new_timer(&timer_config, &gptimer)); // 创建新的定时器

    gptimer_event_callbacks_t cbs = { // 定义定时器事件回调
        .on_alarm = example_timer_on_alarm_cb_v1, // 定义警报回调函数
    };
    ESP_ERROR_CHECK(gptimer_register_event_callbacks(gptimer, &cbs, queue)); // 注册定时器事件回调

    ESP_LOGI(TAG, "Enable timer"); // 启用定时器
    ESP_ERROR_CHECK(gptimer_enable(gptimer)); // 检查定时器是否启用

    ESP_LOGI(TAG, "Start timer, stop it at alarm event"); // 启动定时器，在警报事件时停止
    gptimer_alarm_config_t alarm_config1 = { // 定义警报配置
        .alarm_count = 1000000, // 周期为1s
    };
    ESP_ERROR_CHECK(gptimer_set_alarm_action(gptimer, &alarm_config1)); // 设置警报动作
    ESP_ERROR_CHECK(gptimer_start(gptimer)); // 启动定时器
    if (xQueueReceive(queue, &ele, pdMS_TO_TICKS(2000))) { // 如果在2000ms内接收到队列消息
        ESP_LOGI(TAG, "Timer stopped, count=%llu", ele.event_count); // 打印定时器停止和计数值
    } else {
        ESP_LOGW(TAG, "Missed one count event"); // 打印错过一个计数事件的警告
    }

    ESP_LOGI(TAG, "Set count value"); // 设置计数值
    ESP_ERROR_CHECK(gptimer_set_raw_count(gptimer, 100)); // 设置原始计数值为100
    ESP_LOGI(TAG, "Get count value"); // 获取计数值
    uint64_t count; // 定义计数变量
    ESP_ERROR_CHECK(gptimer_get_raw_count(gptimer, &count)); // 获取原始计数值
    ESP_LOGI(TAG, "Timer count value=%llu", count); // 打印定时器计数值

    // 在更新警报回调之前，我们应确保定时器不处于启用状态
    ESP_LOGI(TAG, "Disable timer"); // 禁用定时器
    ESP_ERROR_CHECK(gptimer_disable(gptimer)); // 检查定时器是否禁用
    // 设置新的回调函数
    cbs.on_alarm = example_timer_on_alarm_cb_v2; // 定义新的警报回调函数
    ESP_ERROR_CHECK(gptimer_register_event_callbacks(gptimer, &cbs, queue)); // 注册新的定时器事件回调
    ESP_LOGI(TAG, "Enable timer"); // 启用定时器
    ESP_ERROR_CHECK(gptimer_enable(gptimer)); // 检查定时器是否启用

    ESP_LOGI(TAG, "Start timer, auto-reload at alarm event"); // 启动定时器，在警报事件时自动重载
    gptimer_alarm_config_t alarm_config2 = { // 定义新的警报配置
        .reload_count = 0, // 重载计数为0
        .alarm_count = 1000000, // 周期为1s
        .flags.auto_reload_on_alarm = true, // 在警报事件时自动重载
    };
    ESP_ERROR_CHECK(gptimer_set_alarm_action(gptimer, &alarm_config2)); // 设置新的警报动作
    ESP_ERROR_CHECK(gptimer_start(gptimer)); // 启动定时器
    int record = 4; // 定义记录变量
    while (record) { // 当记录变量不为0时
        if (xQueueReceive(queue, &ele, pdMS_TO_TICKS(2000))) { // 如果在2000ms内接收到队列消息
            ESP_LOGI(TAG, "Timer reloaded, count=%llu", ele.event_count); // 打印定时器重载和计数值
            record--; // 记录变量减1
        } else {
            ESP_LOGW(TAG, "Missed one count event"); // 打印错过一个计数事件的警告
        }
    }
    ESP_LOGI(TAG, "Stop timer"); // 停止定时器
    ESP_ERROR_CHECK(gptimer_stop(gptimer)); // 检查定时器是否停止

    ESP_LOGI(TAG, "Disable timer"); // 禁用定时器
    ESP_ERROR_CHECK(gptimer_disable(gptimer)); // 检查定时器是否禁用
    cbs.on_alarm = example_timer_on_alarm_cb_v3; // 定义新的警报回调函数
    ESP_ERROR_CHECK(gptimer_register_event_callbacks(gptimer, &cbs, queue)); // 注册新的定时器事件回调
    ESP_LOGI(TAG, "Enable timer"); // 启用定时器
    ESP_ERROR_CHECK(gptimer_enable(gptimer)); // 检查定时器是否启用

    ESP_LOGI(TAG, "Start timer, update alarm value dynamically"); // 启动定时器，动态更新警报值
    gptimer_alarm_config_t alarm_config3 = { // 定义新的警报配置
        .alarm_count = 1000000, // 周期为1s
    };
    ESP_ERROR_CHECK(gptimer_set_alarm_action(gptimer, &alarm_config3)); // 设置新的警报动作
    ESP_ERROR_CHECK(gptimer_start(gptimer)); // 启动定时器
    record = 4; // 重新定义记录变量
    while (record) { // 当记录变量不为0时
        if (xQueueReceive(queue, &ele, pdMS_TO_TICKS(2000))) { // 如果在2000ms内接收到队列消息
            ESP_LOGI(TAG, "Timer alarmed, count=%llu", ele.event_count); // 打印定时器警报和计数值
            record--; // 记录变量减1
        } else {
            ESP_LOGW(TAG, "Missed one count event"); // 打印错过一个计数事件的警告
        }
    }

    ESP_LOGI(TAG, "Stop timer"); // 停止定时器
    ESP_ERROR_CHECK(gptimer_stop(gptimer)); // 检查定时器是否停止
    ESP_LOGI(TAG, "Disable timer"); // 禁用定时器
    ESP_ERROR_CHECK(gptimer_disable(gptimer)); // 检查定时器是否禁用
    ESP_LOGI(TAG, "Delete timer"); // 删除定时器
    ESP_ERROR_CHECK(gptimer_del_timer(gptimer)); // 检查定时器是否删除

    vQueueDelete(queue); // 删除队列
}
