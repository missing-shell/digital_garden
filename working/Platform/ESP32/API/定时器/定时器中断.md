```c++

#include <stdio.h>
#include "driver/gpio.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_timer.h"

#define LED_PIN GPIO_NUM_18

void led_blink(void* arg);

void app_main(void)
{
    gpio_set_level(LED_PIN, 1);  // 设置LED引脚电平为高，使灯泡亮起

    gpio_config_t ioconfig = {
        .pin_bit_mask = (1ULL << LED_PIN),  // LED引脚位掩码
        .mode = GPIO_MODE_OUTPUT  // 输出模式
    };
    gpio_config(&ioconfig);

    // 创建定时器配置
    esp_timer_handle_t timer_handle;
    esp_timer_create_args_t timer_config = {
        .callback = &led_blink,  // 定时器中断回调函数
        .arg = NULL,  // 回调函数参数
        .name = "led_timer"  // 定时器名称
    };

    // 创建定时器
    esp_timer_create(&timer_config, &timer_handle);

    // 启动定时器，并设置闪烁间隔为1秒
    esp_timer_start_periodic(timer_handle, 1000000);

    while (1)
    {
        vTaskDelay(pdMS_TO_TICKS(1000));  // 延迟1秒
    }
}

// LED灯闪烁的函数
void led_blink(void* arg)
{
    static bool led_state = false;
    led_state = !led_state;
    gpio_set_level(LED_PIN, led_state);
}
```
