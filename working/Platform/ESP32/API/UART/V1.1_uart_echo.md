### 配置
* ESP32C6有两个UART控制器，UART0和UART1。这两个UART控制器都可以用于通用的UART任务，如数据发送和接收。
* UART0通常用于打印调试信息，因为它默认连接到芯片的U0TXD和U0RXD引脚，这些引脚通常连接到USB转串口芯片的TX和RX引脚，以便可以通过USB接口与PC通信。
* UART1可以用于其他UART通信任务，如与其他设备的串行通信。
* UART事件服务
	注册UART服务主要是为了使用UART的事件驱动模式。
	在ESP-IDF的UART驱动中，如果你需要使用UART的读写功能，你需要在初始化UART后注册一个UART服务。
	这个函数会为UART端口分配一个队列，用于存储UART事件，如接收到的数据等。
#### 思路
* 使用ESP-IDF框架的UART回声示例。它的主要功能是读取UART接收到的数据，并将其回传。
* 和[[../../淘晶驰串口屏/TJC串口屏]]连接，进行相互通信

```c++
#include <stdio.h>  // Include the standard input/output library
#include <stdarg.h> // Include the standard argument library
#include <string.h>
#include "freertos/FreeRTOS.h" // Include the FreeRTOS library
#include "freertos/task.h"     // Include the FreeRTOS task library
#include "driver/uart.h"       // Include the ESP-IDF UART driver library
#include "driver/gpio.h"       // Include the ESP-IDF GPIO driver library
#include "sdkconfig.h"         // Include the SDK configuration file
#include "esp_log.h"           // Include the ESP-IDF log library
// #include"tjc_usart.h"

/**
 * This is an example which echos any data it receives on configured UART back to the sender,
 * with hardware flow control turned off. It does not use UART driver event queue.
 *
 * - Port: configured UART
 * - Receive (Rx) buffer: on
 * - Transmit (Tx) buffer: off
 * - Flow control: off
 * - Event queue: off
 * - Pin assignment: see defines below (See Kconfig)
 */

#define ECHO_TEST_TXD (10)                 // Define the TXD pin number
#define ECHO_TEST_RXD (11)                 // Define the RXD pin number
#define ECHO_TEST_RTS (UART_PIN_NO_CHANGE) // Define the RTS pin number
#define ECHO_TEST_CTS (UART_PIN_NO_CHANGE) // Define the CTS pin number

#define ECHO_UART_PORT_NUM (UART_NUM_1) // Define the UART port number
#define ECHO_UART_BAUD_RATE (9600)      // Define the UART baud rate
#define ECHO_TASK_STACK_SIZE (2048)     // Define the task stack size

static const char *TAG = "UART TEST"; // Define the log tag

#define BUF_SIZE (1024) // Define the buffer size


static void echo_task(void *arg) // Define the echo task function
{
    /* Configure parameters of an UART driver,
     * communication pins and install the driver */
    uart_config_t uart_config = {
        // Define and initialize the UART configuration
        .baud_rate = ECHO_UART_BAUD_RATE,      // Set the baud rate
        .data_bits = UART_DATA_8_BITS,         // Set the data bits
        .parity = UART_PARITY_DISABLE,         // Disable parity
        .stop_bits = UART_STOP_BITS_1,         // Set the stop bits
        .flow_ctrl = UART_HW_FLOWCTRL_CTS_RTS, // enable hardware flow control
        .source_clk = UART_SCLK_DEFAULT,       // Set the source clock
        .rx_flow_ctrl_thresh = 122,
    };
    int intr_alloc_flags = 0; // Define the interrupt allocation flags

    // #if CONFIG_UART_ISR_IN_IRAM                // If the UART ISR is in IRAM
    //     intr_alloc_flags = ESP_INTR_FLAG_IRAM; // Set the interrupt allocation flags to ESP_INTR_FLAG_IRAM
    // #endif

    ESP_ERROR_CHECK(uart_driver_install(ECHO_UART_PORT_NUM, BUF_SIZE * 2, 0, 0, NULL, intr_alloc_flags));          // Install the UART driver
    ESP_ERROR_CHECK(uart_param_config(ECHO_UART_PORT_NUM, &uart_config));                                          // Configure the UART parameters
    ESP_ERROR_CHECK(uart_set_pin(ECHO_UART_PORT_NUM, ECHO_TEST_TXD, ECHO_TEST_RXD, ECHO_TEST_RTS, ECHO_TEST_CTS)); // Set the UART pins

    // Configure a temporary buffer for the incoming data
    uint8_t *data = (uint8_t *)malloc(BUF_SIZE); // Allocate memory for the data buffer

    // char *str = "\x00";
    char *r_data = "12345";     // Your data
    int r_len = strlen(r_data); // Calculate the length of your data

    

    
    char str_send[110] ;
     //char val_send[] = "n0.val=256\xff\xff\xff";
     float p=3.124;
    //char *s=p;
    snprintf(str_send, sizeof(str_send), "t0.txt=\"%.3f\"\xff\xff\xff",p);
    
    int num = 12222;
    char val_send[100];//不能太大，以免栈溢出
    sprintf(val_send,"n0.val=%d\xff\xff\xff", num);

    int pi=332211314;
    char float_send[50];
    snprintf(float_send, sizeof(float_send), "x0.val=%d\xff\xff\xff", pi);
    uart_write_bytes(ECHO_UART_PORT_NUM, float_send, strlen(float_send));

    uart_write_bytes(ECHO_UART_PORT_NUM, str_send, strlen(str_send));
    uart_write_bytes(ECHO_UART_PORT_NUM, val_send, strlen(val_send));

    ESP_LOGI(TAG, "Send str: %s", (char *)str_send); // Log the received data
    ESP_LOGI(TAG, "Send str: %s", (char *)val_send); // Log the received data
    ESP_ERROR_CHECK(uart_wait_tx_done(ECHO_UART_PORT_NUM, 100));
    while (1)
    {
        // Read data from the UART
        int len = uart_read_bytes(ECHO_UART_PORT_NUM, data, (BUF_SIZE - 1), 20 / portTICK_PERIOD_MS); // Read bytes from the UART
        if (len)
        {                                                      // If data was read
            data[len] = '\0';                                  // Null-terminate the data
            ESP_LOGI(TAG, "Recv       str: %s", (char *)data); // Log the received data
        }
    }
}

void app_main(void) // Define the main application function
{
    xTaskCreate(echo_task, "uart_echo_task", ECHO_TASK_STACK_SIZE, NULL, 10, NULL); // Create the echo task
}
```
