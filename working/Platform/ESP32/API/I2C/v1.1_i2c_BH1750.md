```c++
static esp_err_t i2c_master_sensor_test(i2c_port_t i2c_num, uint8_t *data_h, uint8_t *data_l)
{
    int ret;
    i2c_cmd_handle_t cmd = i2c_cmd_link_create(); // Create an I2C command handle
    i2c_master_start(cmd);                                              // Start signal
    i2c_master_write_byte(cmd, BH1750_SENSOR_ADDR << 1 | WRITE_BIT, ACK_CHECK_EN); // Write slave address + write bit and enable ACK checking
    i2c_master_write_byte(cmd, BH1750_CMD_START, ACK_CHECK_EN);         // Write command to start measurement
    i2c_master_stop(cmd);                                               // Stop signal
    ret = i2c_master_cmd_begin(i2c_num, cmd, 1000 / portTICK_PERIOD_MS); // Execute the I2C command
    i2c_cmd_link_delete(cmd);
    if (ret != ESP_OK) {                                                 // Check if operation was successful
        return ret;
    }
    vTaskDelay(30 / portTICK_PERIOD_MS);                                 // Delay before reading data
    cmd = i2c_cmd_link_create();
    i2c_master_start(cmd);                                              // Start signal
    i2c_master_write_byte(cmd, BH1750_SENSOR_ADDR << 1 | READ_BIT, ACK_CHECK_EN); // Write slave address + read bit and enable ACK checking
    i2c_master_read_byte(cmd, data_h, ACK_VAL);                          // Read MSB data with ACK
    i2c_master_read_byte(cmd, data_l, NACK_VAL);                         // Read LSB data without ACK
    i2c_master_stop(cmd);                                                // Stop signal
    ret = i2c_master_cmd_begin(i2c_num, cmd, 1000 / portTICK_PERIOD_MS);  // Execute the I2C command
    i2c_cmd_link_delete(cmd);
    return ret;
}
```

### i2c master initialization
```c
/**
 * @brief i2c master initialization
 */
static esp_err_t i2c_master_init(void)
{
    int i2c_master_port = I2C_MASTER_NUM; // Define the I2C master port number
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER, // Set the I2C mode as master
        .sda_io_num = I2C_MASTER_SDA_IO, // Set the SDA pin number
        .sda_pullup_en = GPIO_PULLUP_ENABLE, // Enable the pull-up resistor for SDA pin
        .scl_io_num = I2C_MASTER_SCL_IO, // Set the SCL pin number
        .scl_pullup_en = GPIO_PULLUP_ENABLE, // Enable the pull-up resistor for SCL pin
        .master.clk_speed = I2C_MASTER_FREQ_HZ, // Set the clock speed for I2C communication
        // .clk_flags = 0,          /*!< Optional, you can use I2C_SCLK_SRC_FLAG_* flags to choose i2c source clock here. */
    };
    esp_err_t err = i2c_param_config(i2c_master_port, &conf); // Configure I2C parameters
    if (err != ESP_OK) { // Check if configuration was successful
        return err;
    }
    return i2c_driver_install(i2c_master_port, conf.mode, I2C_MASTER_RX_BUF_DISABLE, I2C_MASTER_TX_BUF_DISABLE, 0); // Install I2C driver
}
```

```c++
        xSemaphoreTake(print_mux, portMAX_DELAY);
        if (ret == ESP_ERR_TIMEOUT) {
            ESP_LOGE(TAG, "I2C Timeout");                                    // Print error message if I2C timeout occurs
        } else if (ret == ESP_OK) {
            printf("*******************\n");
            printf("TASK[%d]  MASTER READ SENSOR( BH1750 )\n", task_idx);     // Print sensor read operation information
            printf("*******************\n");
            printf("data_h: %02x\n", sensor_data_h);                         // Print MSB data
            printf("data_l: %02x\n", sensor_data_l);                         // Print LSB data
            printf("sensor val: %.02f [Lux]\n", (sensor_data_h << 8 | sensor_data_l) / 1.2); // Calculate and print Lux value
        } else {
            ESP_LOGW(TAG, "%s: No ack, sensor not connected...skip...", esp_err_to_name(ret)); // Print warning message if no acknowledgement is received
        }
        xSemaphoreGive(print_mux);
        vTaskDelay((DELAY_TIME_BETWEEN_ITEMS_MS * (task_idx + 1)) / portTICK_PERIOD_MS); // Delay between different test items
        //---------------------------------------------------
    }
    vSemaphoreDelete(print_mux);
    vTaskDelete(NULL);
}
```

### 信号量
```markdown 
1. 在 `app_main` 函数中，使用 `xSemaphoreCreateMutex` 函数创建了一个互斥信号量 `print_mux`。互斥信号量是一种特殊类型的信号量，它只有两个值（0和1），用于保护对共享资源的访问。
    
2. 在 `i2c_test_task` 函数中，每次读取传感器数据并打印之前，都会先使用 `xSemaphoreTake` 函数尝试获取 `print_mux` 信号量。如果信号量可用（即，没有其他任务正在使用打印资源），则 `xSemaphoreTake` 函数会立即返回，并且该任务可以安全地进行打印操作。如果信号量不可用（即，其他任务正在使用打印资源），则 `xSemaphoreTake` 函数会阻塞，直到信号量变为可用。
    
3. 打印操作完成后，使用 `xSemaphoreGive` 函数释放 `print_mux` 信号量，使其他可能正在等待的任务可以获取信号量并访问打印资源。
    
4. 在 `i2c_test_task` 函数的最后，使用 `vSemaphoreDelete` 函数删除 `print_mux` 信号量。这是一个好习惯，可以避免资源泄漏。

5. `SemaphoreHandle_t print_mux = NULL;` 这行代码定义了一个全局变量 `print_mux`，它在整个程序中都是可见的，以确保可以在任何函数中使用。

6. 总的来说，信号量在这段代码中的作用是同步和保护对共享打印资源的访问，确保在任何时候只有一个任务可以打印，从而避免了可能的数据混乱。
```
