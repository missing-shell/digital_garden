#include <stdio.h>
#include "esp_log.h"
#include "driver/i2c.h"
#include "sdkconfig.h"
#include <math.h>
static const char *TAG = "V1.1_i2c_MCP4725";

#define I2C_MASTER_PORT_NUM 0

#define DATA_LENGTH 512                  /*!< Data buffer length of test buffer */
#define RW_TEST_LENGTH 128               /*!< Data length for r/w test, [0,DATA_LENGTH] */
#define DELAY_TIME_BETWEEN_ITEMS_MS 1000 /*!< delay time between different test items */

#define I2C_MASTER_SCL_IO GPIO_NUM_7       /*!< gpio number for I2C master clock */
#define I2C_MASTER_SDA_IO GPIO_NUM_6       /*!< gpio number for I2C master data  */
#define I2C_MASTER_NUM I2C_MASTER_PORT_NUM /*!< I2C port number for master dev */
#define I2C_MASTER_FREQ_HZ 100000          /*!< I2C master clock frequency */
#define I2C_MASTER_TX_BUF_DISABLE 0        /*!< I2C master doesn't need buffer */
#define I2C_MASTER_RX_BUF_DISABLE 0        /*!< I2C master doesn't need buffer */


#define WRITE_BIT I2C_MASTER_WRITE /*!< I2C master write */
#define READ_BIT I2C_MASTER_READ   /*!< I2C master read */
#define ACK_CHECK_EN 0x1           /*!< I2C master will check ack from slave */
#define ACK_CHECK_DIS 0x0          /*!< I2C master will not check ack from slave */
#define ACK_VAL 0x0                /*!< I2C ack value */
#define NACK_VAL 0x1               /*!< I2C nack value */

#define PI 3.14159265
#define DAC_MAX 4095
// SemaphoreHandle_t print_mux = NULL; // Semaphore handle for controlling access to shared printing resource

// Function to write a DAC value to the MCP4725
static esp_err_t i2c_master_mcp4725_write(i2c_port_t i2c_num, uint16_t dac_value)
{
    int ret;
    i2c_cmd_handle_t cmd = i2c_cmd_link_create();                                     // Create an I2C command handle
    i2c_master_start(cmd);                                                            // Start signal                                                           // Start signal
    i2c_master_write_byte(cmd, (MCP4725_SENSOR_ADDR << 1) | WRITE_BIT, ACK_CHECK_EN); // Write slave address + write bit and enable ACK checking
    i2c_master_write_byte(cmd, MCP4725_CMD_WRITE, ACK_CHECK_EN);                      // Write the write command
    i2c_master_write_byte(cmd, dac_value >> 4, ACK_CHECK_EN);                         // Write the high byte of the DAC value
    i2c_master_write_byte(cmd, dac_value << 4, ACK_CHECK_EN);                         // Write the low byte of the DAC value
    i2c_master_stop(cmd);                                                             // Stop signal
    ret = i2c_master_cmd_begin(i2c_num, cmd, 1000 / portTICK_PERIOD_MS);              // Execute the I2C command
    i2c_cmd_link_delete(cmd);                                                         // Delete the I2C command handle
    return ret;
}

/**
 * @brief i2c master initialization
 */
static esp_err_t i2c_master_init(void)
{
    int i2c_master_port = I2C_MASTER_NUM; // Define the I2C master port number
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER,                // Set the I2C mode as master
        .sda_io_num = I2C_MASTER_SDA_IO,        // Set the SDA pin number
        .sda_pullup_en = GPIO_PULLUP_ENABLE,    // Enable the pull-up resistor for SDA pin
        .scl_io_num = I2C_MASTER_SCL_IO,        // Set the SCL pin number
        .scl_pullup_en = GPIO_PULLUP_ENABLE,    // Enable the pull-up resistor for SCL pin
        .master.clk_speed = I2C_MASTER_FREQ_HZ, // Set the clock speed for I2C communication
        // .clk_flags = 0,          /*!< Optional, you can use I2C_SCLK_SRC_FLAG_* flags to choose i2c source clock here. */
    };
    esp_err_t err = i2c_param_config(i2c_master_port, &conf); // Configure I2C parameters
    if (err != ESP_OK)
    { // Check if configuration was successful
        return err;
    }
    return i2c_driver_install(i2c_master_port, conf.mode, I2C_MASTER_RX_BUF_DISABLE, I2C_MASTER_TX_BUF_DISABLE, 0); // Install I2C driver
}

/**
 * @brief test function to show buffer
 */
static void i2c_test_task(void *arg)
{
    int ret;
    // int task_idx = (int)arg;
    double angle = 0.0;
    while (1)
    {
        // Calculate the value of the sine wave
        double sin_val = (sin(angle) + 1.0) / 2.0;          // Shift range from -1~1 to 0~1
        uint16_t dac_value = (uint16_t)(sin_val * DAC_MAX+0.5); // Scale to range 0~4095

        // Write the DAC value
        ret = i2c_master_mcp4725_write(I2C_MASTER_NUM, dac_value);

        if (ret != ESP_OK) // Check if the operation was successful
        {
            ESP_LOGE(TAG, "MCP4725 Write Failed"); // Print an error message if the operation failed
        }
        else
        {
            ESP_LOGI(TAG, "MCP4725 DAC Value: %d", dac_value); // Print the DAC value if the operation was successful
        }
        // Increase the angle
        angle += 0.01;
        if (angle > 2 * PI)
        {
            angle -= 2 * PI;
        }

        vTaskDelay(10 / portTICK_PERIOD_MS);
    }
}

void app_main(void)
{
    // print_mux = xSemaphoreCreateMutex(); // Create a semaphore for controlling access to shared printing resource
    ESP_ERROR_CHECK(i2c_master_init()); // Initialize I2C master

    /*Test I2C communication in parallel. Each task independently communicates with the BH1750 light intensity sensor, reads the data and prints the results. */
    xTaskCreate(i2c_test_task, "i2c_test_task_0", 1024 * 2, (void *)0, 10, NULL); // Create first I2C test task
    // xTaskCreate(i2c_test_task, "i2c_test_task_1", 1024 * 2, (void *)1, 10, NULL); // Create second I2C test task
}

### tips

#### 波形锯齿状
* 在MCP4725 DAC中使用0x40模式（快速模式）时，DAC的更新速率主要取决于I2C总线的速度和单个I2C事务的执行时间。在最大400kHz的I2C速度下，理论上MCP4725可以达到大约15kHz的更新速率。
* 实际的更新速率可能会受到多个因素的影响，包括I2C总线的速度、处理器的性能、其他任务的CPU使用率、以及操作系统的调度策略。如果这些因素使得实际的更新速率低于期望的波形频率，那么就可能会看到锯齿状的波形。
* vTaskDelay`函数用于在每次更新DAC值后引入一个延迟。这个延迟时间应该尽可能地小，以达到更高的更新频率。然而，这个延迟时间不能为0，因为那样会使得CPU使用率达到100%，并可能导致操作系统无法执行其他任务。
* 过小的延迟可能会导致CPU过载，而过大的延迟可能会导致波形不够平滑。因此，找到合适的延迟值可能需要一些试验。
#### 周期
* 周期是由两个因素决定的：角度的增量和任务延迟。

1. 角度的增量：每次循环，角度都会增加一个固定的值（在代码中是0.01）。这个增量决定了完成一个完整周期需要多少次循环。例如，如果每次循环角度增加0.01，那么需要大约628次循环（因为2π约等于6.28）来完成一个周期。
2. 任务延迟：`vTaskDelay`函数会在每次循环后引入一个延迟。这个延迟决定了每次循环的时间。例如，如果延迟是100毫秒，那么每次循环就需要100毫秒的时间。
3. 因此，一个周期的时间就是循环次数乘以每次循环的时间。
