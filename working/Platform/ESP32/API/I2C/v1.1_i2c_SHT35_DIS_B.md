static esp_err_t i2c_master_sensor_test(i2c_port_t i2c_num, uint8_t *data)
{
    int ret;
    i2c_cmd_handle_t cmd = i2c_cmd_link_create(); // Create an I2C command handle
    // 1. Set measurement mode
    // _________________________________________________________________
    // | start | slave_addr + wr_bit + ack | write 2 bytes + ack  | stop |
    // --------|---------------------------|---------------------|------|
    i2c_master_start(cmd);                                                              // Start signal
    i2c_master_write_byte(cmd, SHT35_DIS_B_SENSOR_ADDR << 1 | WRITE_BIT, ACK_CHECK_EN); // Write slave address + write bit and enable ACK checking
    i2c_master_write_byte(cmd, SHT35_DIS_B_CMD_START >> 8, ACK_CHECK_EN);               // write high byte of command
    i2c_master_write_byte(cmd, SHT35_DIS_B_CMD_START & 0xFF, ACK_CHECK_EN);             // write low byte of command
    i2c_master_stop(cmd);                                                               // Stop signal
    ret = i2c_master_cmd_begin(i2c_num, cmd, 1000 / portTICK_PERIOD_MS);                // Execute the I2C command
    i2c_cmd_link_delete(cmd);
    if (ret != ESP_OK)
    { // Check if operation was successful
        return ret;
    }
    // 2. Wait for measurement to complete (the exact time depends on the measurement mode)
    vTaskDelay(50 / portTICK_PERIOD_MS); // Delay before reading data
    cmd = i2c_cmd_link_create();
    // 3. Read measurement data
    // ______________________________________________________________________________________
    // | start | slave_addr + rd_bit + ack | read 2 bytes + ack (temperature) | read 2 bytes + ack (humidity) | stop |
    // --------|---------------------------|--------------------|--------------------|------|
    i2c_master_start(cmd);                                                             // Start signal
    i2c_master_write_byte(cmd, SHT35_DIS_B_SENSOR_ADDR << 1 | READ_BIT, ACK_CHECK_EN); // Write slave address + read bit and enable ACK checking

    i2c_master_read_byte(cmd, &data[0], ACK_VAL);                        // Read first byte with ACK
    i2c_master_read_byte(cmd, &data[1], ACK_VAL);                        // Read second byte with ACK
    i2c_master_read_byte(cmd, &data[2], ACK_VAL);                        // Read CRC for temperature
    i2c_master_read_byte(cmd, &data[3], ACK_VAL);                        // Read third byte with ACK
    i2c_master_read_byte(cmd, &data[4], ACK_VAL);                        // Read fourth byte with ACK
    i2c_master_read_byte(cmd, &data[5], NACK_VAL);                       // Read CRC for humidity
    i2c_master_stop(cmd);                                                // Stop signal
    ret = i2c_master_cmd_begin(i2c_num, cmd, 1000 / portTICK_PERIOD_MS); // Execute the I2C command
    i2c_cmd_link_delete(cmd);
    return ret;
}

/**
 * @brief i2c master initialization
 */
static esp_err_t i2c_master_init(void)
{
    int i2c_master_port = I2C_MASTER_NUM; // Define the I2C master port number
    i2c_config_t conf = {
        .mode = I2C_MODE_MASTER,                // Set the I2C mode as master
        .sda_io_num = I2C_MASTER_SDA_IO,        // Set the SDA pin number
        .sda_pullup_en = GPIO_PULLUP_ENABLE,    // Enable the pull-up resistor for SDA pin
        .scl_io_num = I2C_MASTER_SCL_IO,        // Set the SCL pin number
        .scl_pullup_en = GPIO_PULLUP_ENABLE,    // Enable the pull-up resistor for SCL pin
        .master.clk_speed = I2C_MASTER_FREQ_HZ, // Set the clock speed for I2C communication
        // .clk_flags = 0,          /*!< Optional, you can use I2C_SCLK_SRC_FLAG_* flags to choose i2c source clock here. */
    };
    esp_err_t err = i2c_param_config(i2c_master_port, &conf); // Configure I2C parameters
    if (err != ESP_OK)
    { // Check if configuration was successful
        return err;
    }
    return i2c_driver_install(i2c_master_port, conf.mode, I2C_MASTER_RX_BUF_DISABLE, I2C_MASTER_TX_BUF_DISABLE, 0); // Install I2C driver
}

/**
 * @brief Calculate the CRC for the sensor data
 *
 * This function calculates the CRC (Cyclic Redundancy Check) for the sensor data.
 * The CRC is a type of hash function used to produce a checksum, which is a small-sized datum
 * derived from the original data and used to detect errors during transmission or storage.
 * The SHT35-DIS-B sensor sends a CRC after each data transmission, which can be used to verify the integrity of the data.
 *
 * @param data The data for which to calculate the CRC
 * @param len The length of the data
 * @return The calculated CRC
 */
uint8_t calculate_crc(uint8_t *data, uint8_t len)
{
    // CRC initial value
    uint8_t crc = 0xFF;

    // Calculate 8-Bit checksum with the given polynomial
    for (uint8_t byteCtr = 0; byteCtr < len; byteCtr++)
    {
        crc ^= (data[byteCtr]);
        for (uint8_t bit = 8; bit > 0; --bit)
        {
            if (crc & 0x80)
            {
                // If the leftmost (most significant) bit is '1', take it out (via XOR with a shifted mask)
                // and replace it with the remainder of the division (via XOR) of the polynomial
                crc = (crc << 1) ^ 0x131;
            }
            else
            {
                // If the leftmost bit is '0', just shift the register contents
                crc = (crc << 1);
            }
        }
    }
    return crc;
}

        //---------------------------------------------------
    }

}

### 模式
#### Single Shot Mode

- **单次测量模式**：在这种模式下，传感器只在接收到测量命令时进行一次测量。这种模式适用于设备的功耗需要尽可能低，或者环境条件变化不大，不需要频繁测量的情况。在此模式下，每次测量后，传感器会自动回到低功耗的睡眠状态。
#### Periodic Mode

- **周期测量模式**：在这种模式下，传感器会根据设定的频率自动进行测量，并将数据存储在内部的缓冲区中，可以随时读取。这种模式适用于需要连续监控环境条件，或者环境条件变化较大，需要频繁测量的情况。在此模式下，传感器会持续地进行测量和更新数据，消耗的功耗相对较高。

### CRC
- 6个字节的数据包含了温度和湿度的原始数据以及用于校验数据正确性的CRC校验码。具体来说，前两个字节是温度数据，第三个字节是温度数据的CRC校验码，第四和第五个字节是湿度数据，第六个字节是湿度数据的CRC校验码。
