### (简单的状态机来处理按键) 软件消抖

#### code
```c++
#define BUTTON_PIN  GPIO_NUM_12
#define DEBOUNCE_DELAY_MS  10

typedef enum {
  BUTTON_STATE_IDLE,      // 按键空闲状态
  BUTTON_STATE_PRESSED,   // 按键按下状态
  BUTTON_STATE_RELEASED   // 按键释放状态
} ButtonState;

void button_task(void *pvParameters) {
  ButtonState state = BUTTON_STATE_IDLE;   // 初始化按键状态为空闲状态
  TickType_t lastStateChangeTime = 0;      // 记录状态改变的时间

  while (1) {
    // 读取按键状态
    int buttonValue = gpio_get_level(BUTTON_PIN);

    // 根据当前状态进行处理
    switch (state) {
      case BUTTON_STATE_IDLE:
        if (buttonValue == 0) {
          // 按键被按下
          state = BUTTON_STATE_PRESSED;
          lastStateChangeTime = xTaskGetTickCount();
        }
        break;

      case BUTTON_STATE_PRESSED:
        if (buttonValue == 1) {
          // 按键被释放
          state = BUTTON_STATE_RELEASED;
          lastStateChangeTime = xTaskGetTickCount();
        }
        break;

      case BUTTON_STATE_RELEASED:
        if (buttonValue == 0) {
          // 按键再次被按下
          state = BUTTON_STATE_PRESSED;
          lastStateChangeTime = xTaskGetTickCount();
        }
        break;
    }

    // 检查是否达到消抖延迟
    TickType_t currentTime = xTaskGetTickCount();
    TickType_t elapsedTime = currentTime - lastStateChangeTime;
    if (elapsedTime >= pdMS_TO_TICKS(DEBOUNCE_DELAY_MS)) {
      // 消抖延迟已达到，执行相应的操作
      if (state == BUTTON_STATE_PRESSED) {
        // 按键被按下的操作
        printf("按键按下\n");
      } else if (state == BUTTON_STATE_RELEASED) {
        // 按键被释放的操作
        printf("按键释放\n");
      }

      // 重置状态为按键空闲状态
      state = BUTTON_STATE_IDLE;
    }

    // 延时一段时间，避免过多的 CPU 占用
    vTaskDelay(pdMS_TO_TICKS(1));
  }
}

void app_main() {
  // 配置按键引脚为输入模式
  gpio_config_t buttonConfig = {
    .pin_bit_mask = (1ULL << BUTTON_PIN),
    .mode = GPIO_MODE_INPUT,
    .pull_up_en = GPIO_PULLUP_ENABLE,
    .intr_type = GPIO_INTR_DISABLE
  };
  gpio_config(&buttonConfig);

  // 创建按键任务
  xTaskCreate(button_task, "button_task", 2048, NULL, 10, NULL);
}
```
