* KEY1 GPIO3-ESP32C6 IO3
* KEY2 GPIO4-ESP32C6 IO21
将GPIO4连接到ESP32C6的引脚21，并将其配置为INPUT_PULLUP模式。这使得引脚上有一个上拉电阻，当开关未按下时，引脚将保持高电平。

* LED1 GPIO1-ESP32C6 IO0
* LED2 GPIO2-ESP32C6 IO1
```c++
#include <stdio.h>
#include "driver/gpio.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define LED_PIN_1 GPIO_NUM_0
#define LED_PIN_2 GPIO_NUM_1
#define BUTTON_PIN_1 GPIO_NUM_3
#define BUTTON_PIN_2 GPIO_NUM_21

volatile int g_led_state_1 = 0; // LED1 的状态，1 表示灭，0 表示亮
volatile int g_led_state_2 = 0; // LED2 的状态，1 表示灭，0 表示亮

static void button_isr_handler_1(void *arg);
static void button_isr_handler_2(void *arg);

void app_main(void)
{
    /*初始化灯泡，低电平*/
    gpio_reset_pin(LED_PIN_1);
    gpio_reset_pin(LED_PIN_2);

    /*安装GPIO中断服务*/
    gpio_install_isr_service(0);

    /*配置按键1引脚*/
    gpio_config_t button1_config = {
        .pin_bit_mask = (1ULL << BUTTON_PIN_1),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .intr_type = GPIO_INTR_POSEDGE,
    };
    gpio_config(&button1_config);

    /*配置按键2引脚*/
    gpio_config_t button2_config = {
        .pin_bit_mask = (1ULL << BUTTON_PIN_2),
        .mode = GPIO_MODE_INPUT,
        .pull_up_en = GPIO_PULLUP_ENABLE,
        .intr_type = GPIO_INTR_POSEDGE,
    };
    gpio_config(&button2_config);

    /*配置LED引脚*/
    gpio_config_t io1config = {
        .pin_bit_mask = (1ULL << LED_PIN_1) | (1ULL << LED_PIN_2),
        .mode = GPIO_MODE_OUTPUT};
    gpio_config(&io1config);

    /*注册按键中断处理函数*/
    gpio_isr_handler_add(BUTTON_PIN_1, button_isr_handler_1, NULL);
    gpio_isr_handler_add(BUTTON_PIN_2, button_isr_handler_2, NULL);

    while (1)
    {
        /*处理中断任务*/
        gpio_set_level(LED_PIN_1, g_led_state_1);
        gpio_set_level(LED_PIN_2, g_led_state_2);
        vTaskDelay(pdMS_TO_TICKS(190)); // 延迟0.19秒，消除抖动
    }
}

void IRAM_ ATTR button_isr_handler_1(void *arg)
{
    g_led_state_1 = !g_led_state_1;
}
void IRAM_ATTR button_isr_handler_2(void *arg)
{
    g_led_state_2 = !g_led_state_2;
}
```
