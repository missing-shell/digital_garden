public class IncomeTaxCalculator {

    // 综合所得税计算方法（考虑年终奖合并）
    private double calculateTaxWithBonus(double income, int specialDeduction, double bonus) {
        double taxableIncome = income - 12 * (specialDeduction + 5000) - bonus;
        double tax = calculateTax(taxableIncome, INCOME_TAX_BRACKETS);
        double bonusTax = calculateBonusTax(bonus, BONUS_TAX_BRACKETS);
        return tax + bonusTax;
    }

    // 综合所得税计算方法（不考虑年终奖）
    private double calculateTaxWithoutBonus(double income, int specialDeduction) {
        double taxableIncome = income - 12 * (specialDeduction + 5000);
        return calculateTax(taxableIncome, INCOME_TAX_BRACKETS);
    }

    // 计算年终奖单独的税
    private double calculateBonusTax(double bonus) {
        double monthlyBonus = bonus / 12;
        return calculateTax(monthlyBonus, BONUS_TAX_BRACKETS) * bonus;
    }

    // 核心税费计算逻辑
    private double calculateTax(double income, int[][] brackets) {
        double tax = 0;
        for (int i = 0; i < brackets.length && income > 0; i++) {
            int upperLimit = brackets[i][0];
            double rate = brackets[i][1] / 100.0;
            if (income > upperLimit) {
                tax += (upperLimit * rate - (i > 0 ? brackets[i-1][0] * brackets[i-1][1] / 100.0 : 0));
                income -= upperLimit;
            } else {
                tax += (income * rate - (i > 0 ? brackets[i-1][0] * brackets[i-1][1] / 100.0 : 0));
                break;
            }
        }
        return tax;
    }

    // 总所得税计算及最优策略选择
    void computeTax(double income, int specialDeduction, double bonus) {
        double taxWithBonus = calculateTaxWithBonus(income, specialDeduction, bonus);
        double taxWithoutBonus = calculateTaxWithoutBonus(income, specialDeduction);

        if (taxWithBonus < taxWithoutBonus) {
            System.out.println("采用奖金单独计算纳税更划算，纳税额为 " + taxWithBonus);
        } else {
            System.out.println("采用奖金合并计算纳税更划算，纳税额为 " + taxWithoutBonus);
        }
    }

    // 定义税率表
    private static final int[][] INCOME_TAX_BRACKETS = {
        {36000, 3}, {144000, 10}, {300000, 20}, {420000, 25}, 
        {660000, 30}, {960000, 35}, {Integer.MAX_VALUE, 45}
    };
    private static final int[][] BONUS_TAX_BRACKETS = {
        {3000, 3}, {12000, 10}, {25000, 20}, {35000, 25}, 
        {55000, 30}, {80000, 35}, {Integer.MAX_VALUE, 45}
    };
}
