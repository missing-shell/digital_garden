public class ManchesterEncoding {

    /**
     * 实现曼彻斯特编码
     * @param input 8位二进制数的字符串，例如："10101010"
     * @return 编码后的字符串，包含起始位10和结束位00，以及原始数据的曼彻斯特编码
     */
    public static String encode(String input) {
        // 添加起始位
        StringBuilder encoded = new StringBuilder("10");
        
        // 对输入的每一位进行编码
        for (int i = 0; i < input.length(); i++) {
            char bit = input.charAt(i);
            if (bit == '1') {
                // 逻辑1编码为10
                encoded.append("10");
            } else if (bit == '0') {
                // 逻辑0编码为01
                encoded.append("01");
            }
        }
        
        // 添加结束位
        encoded.append("00");
        
        // 进行偶校验，确保编码后的字符串中1的个数为偶数
        int onesCount = countOnes(encoded.toString());
        if (onesCount % 2 != 0) {
            // 如果1的个数为奇数，将最后一个编码位翻转以保证偶校验通过
            if (encoded.charAt(encoded.length() - 2) == '1') {
                encoded.setCharAt(encoded.length() - 2, '0');
            } else {
                encoded.setCharAt(encoded.length() - 2, '1');
            }
        }
        
        return encoded.toString();
    }

    /**
     * 计算字符串中'1'的个数
     * @param str 字符串
     * @return 字符串中'1'的数量
     */
    private static int countOnes(String str) {
        int count = 0;
        for (char c : str.toCharArray()) {
            if (c == '1') {
                count++;
            }
        }
        return count;
    }

    public static void main(String[] args) {
        String binaryData = "10101010"; // 假设这是需要编码的8位二进制数据
        String encodedData = encode(binaryData);
        System.out.println("原始二进制数据: " + binaryData);
        System.out.println("曼彻斯特编码结果: " + encodedData);
    }
}
