## 检查日期合法性
```c++
int months[13] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

bool check(int date)
{
    int year = date / 10000;
    int month = date % 10000 / 100;
    int day = date % 100;

    if (!month || month >= 13 || !day)
    {
        return false;
    }
    if (month != 2 && day > months[month])
    {
        return false;
    }
    if (month == 2)
    {
        bool leap = year % 4 == 0 && year % 100 || year % 400 == 0;
        if (day > 28 + leap)
        {
            return false;
        }
    }
    return true;
}
```
