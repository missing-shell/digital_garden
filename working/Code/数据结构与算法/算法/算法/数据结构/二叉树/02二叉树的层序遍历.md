## 思路
> 层序遍历一个二叉树。就是从左到右一层一层的去遍历二叉树。

- 需要借用一个辅助数据结构即队列来实现，**队列**先进先出，符合一层一层遍历的逻辑
### 模板
#### 迭代
```c++
class Solution{
public:
	vector<vector<int>> leverOrder(TreeNode* root)
	{
		queue<TreeNode*>que;
		if(root!=nullptr)que.push(root);
		vector<vector<int>> result;
		while(!que.empty())
		{
			int size=que.size();//使用固定大小size，因为que.size是不断变化的
			vector<int> vec;
			for(int i=0;i<size;i++)
			{
				TreeNode* node=que.front;
				que.pop();
				vec.push_back(node->val);
				if(node->left)que.push(node->left);
				if(node->right)que.push(node->right);
			}
			result.push_back(vec);
		}
		return result;
	}
};
```
#### 递归
```c++
/**
 * @class Solution
 * @brief 解决二叉树层次遍历问题的类。
 */
class Solution {
public:

    /**
     * @brief 递归函数，实现二叉树的层次遍历。
     * 
     * 此函数通过深度优先搜索的方式，将二叉树节点按层次顺序添加到结果中。
     *
     * @param cur 当前处理的树节点。
     * @param result 保存层次遍历结果的二维向量。
     * @param depth 当前节点所在的层次，用于追踪遍历的深度。
     */
    void order(TreeNode* cur, std::vector<std::vector<int>>& result, int depth) {
        if (cur == nullptr) return; ///< 当前节点为空时，结束递归。

        // 如果当前层次尚未记录，则创建新层次。
        if (result.size() == depth) result.push_back(std::vector<int>());

        result[depth].push_back(cur->val); ///< 将当前节点值添加到对应层次的列表中。

        order(cur->left, result, depth + 1); ///< 递归遍历左子树。
        order(cur->right, result, depth + 1); ///< 递归遍历右子树。
    }

    /**
     * @brief 返回二叉树层次遍历的结果。
     *
     * @param root 二叉树的根节点。
     * @return std::vector<std::vector<int>> 包含层次遍历顺序的二维向量。
     */
    std::vector<std::vector<int>> levelOrder(TreeNode* root) {
        std::vector<std::vector<int>> result; ///< 存储层次遍历结果的容器。
        int depth = 0; ///< 初始化层次为0。

        order(root, result, depth); ///< 调用递归函数开始层次遍历。

        return result; ///< 返回层次遍历的结果。
    }
};
```
