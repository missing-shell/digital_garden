```c++
 #include<iostream>
#include<cstring>

using namespace std;

const int N = 1e5 + 10;

int head, e[N], ne[N],idx;
void init() //初始化
{
	head = -1;
	idx = 0;
}
void add_head(int x) //在头结点处插入
{
	e[idx] = x;
	ne[idx] = head;
	head = idx++;
}
void add_k(int k, int x)//在第K个结点后插入
{
	e[idx] = x;
	ne[idx] = ne[k];
	ne[k] = idx++;
}
void remove(int k)//删除第K个结点
{
	ne[k] = ne[ne[k]];
}
int main()
{
	int t,k,x;
	cin >> t;
	init();
	while (t--) {
		char c;
		cin >> c;
		if (c == 'H') {
			cin >> x;
			add_head(x);
		}
		else if (c == 'D') {
			cin >> k;
			if (!k)head = ne[head];
			else remove(k - 1);
		}
		else if (c == 'I') {
			cin >> k >> x;
			add_k(k-1, x);
		}
	}
	for (int i = head; i != -1; i = ne[i]) {
		cout << e[i] << ' ';
	}
	cout << endl;
	return 0;
}
```
