 #include<iostream>
#include<cmath>
using namespace std;
const int MAXN = 150;
const double eps = 1e-6;
int n;
double a[MAXN][MAXN];
void out()
{
	for (int r = 0; r < n; r++) {
		for (int c = 0; c <= n; c++) {
			printf("%12.2f", a[r][c]);
		}
		printf("\n");
	}
}
int gauss()
{
	int c, r;//c代表枚举的列，r代表枚举的行
	for (c = 0, r = 0; c < n; c++) {
		int t = r;
		for (int i = r; i < n; i++) {
			if (fabs(a[t][c] < fabs(a[i][c])))  t = i;//定位当前列绝对值最大的行
		}
		if (fabs(a[t][c]) < eps) continue;
		for (int y = c; y <= n; y++)swap(a[t][y], a[r][y]);//将该行换到最上边
		for (int y = n; y >= c; y--)a[r][y] /= a[r][c];//将该行首位化为一
		for (int x = r + 1; x < n; x++)
			if (fabs(a[x][c]) > eps) {
				for (int y = n; y >= c; y--)
					a[x][y] -= a[x][c] * a[r][y];//将该列下边所有行的当前行化为零
			}
		//printf("第%d次\n",c + 1);
        //out();
		r++;
	}
	for (int x = n - 1; x >= 0; x--) {//回代求解
		for (int y = x + 1; y < n; y++) {
			a[x][n] -= a[x][y] * a[y][n];
		}
	}
	if (r < n)
	{
		for (int i = r; i < n; i++)
		{
			if (fabs(a[i][n]) > eps)
				return -1;//无解
		}
		return 0;//有无穷多组解
	}

	return 1;//有唯一解
}

int main()
{
	cin >> n;
	for (int i = 0; i < n; i++)
		for (int j = 0; j <= n; j++)
			cin >> a[i][j];
	int t = gauss();
	if (t == 1) {
		for (int i = 0; i < n; i++)printf("x%d=%.2f\n", i + 1, a[i][n]);
	}
	else  cout << t << endl;
	return 0;
}
