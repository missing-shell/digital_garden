非常巧妙，特别是ans[i-1]=ans[i]-ans[i-1];
```c++
#include<iostream>
#include<algorithm>
using namespace std;
int ans[10000010];
int main()
{
	int i,j=0, n;
	cin >> n;
	cin >> ans[0];
	for (i = 1; i < n; i++) {
		cin >> ans[i];
		ans[i - 1] = abs(ans[i] - ans[i - 1]);
	}
	for (i = 1; i < n && j != n - 1; i++) {
		for (j = 0; j < n - 1; j++) {
			if (ans[j] == i)
				break;
		}
	}
	if (i == n)printf("Jolly\n");
	else printf("Not jolly\n");
	return 0;
}
