bool cmp(string a, string b) {
	int  i = 0;
	long long int  x = a[i] - '0', y = b[i] - '0';
	while (x == y && i < min(a.size(), b.size())) {
		i++;
	    x = a[i] - '0', y = b[i] - '0';
	    }
	return x > y;
}```
