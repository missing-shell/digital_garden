 #include<iostream>
#include<algorithm>
#include<cstring>

using namespace std;

const int N = 100;

int ans[N],sum=0,leap=3;
int n ;
bool st[N], dg[2 * N], udg[2 * N];

void dfs(int cnt)
{
	if (cnt > n) {
		sum++;
		if (leap>0) {//此处不可用leap--或leap表示。leap-1之后再次递归到此处leap已经小于零，限制失效
			for (int k = 1; k <= n; k++) {
				printf("%d%c", ans[k], k == n ? '\n' : ' '); 
			}
			leap--;
		}
		/*
		if (sum <= 2) {
			for (int k = 1; k <= n; k++) {
				printf("%d%c", ans[k], k == n ? '\n' : ' ');
			}
			sum++;
		}*/    //更优解，还减少了变量的使用
		return;
	}
	for (int i = 1; i <= n; i++) {
		if (!st[i] && !dg[i + cnt] && !udg[i - cnt + n]) {
			ans[cnt] = i ;//记录当前行(cnt)在第i列
			st[i] = dg[i + cnt] = udg[i - cnt + n] = true;
			dfs(cnt + 1);
			st[i] = dg[i + cnt] = udg[i - cnt + n] = false;
			ans[cnt] = 0;
		}
	}
}
int main()
{
	cin >> n;
	dfs(1);
	printf("%d\n", sum);
	return 0;
}
